```bash
apt install curl -y

curl "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o "awscliv2.zip"
unzip awscliv2.zip
sudo ./aws/install

aws configure

aws sts get-caller-identity

curl -L https://github.com/vmware-tanzu/velero/releases/download/v1.17.2/velero-v1.17.2-linux-amd64.tar.gz \
| tar -xz && sudo mv velero-v1.17.2-linux-amd64/velero /usr/local/bin/ && sudo chmod +x /usr/local/bin/velero

BUCKET=<YOUR_BUCKET>
REGION=<YOUR_REGION>
aws s3api create-bucket \
    --bucket $BUCKET \
    --region $REGION \
    --create-bucket-configuration LocationConstraint=$REGION

aws s3 ls

#Enable versioning
aws s3api put-bucket-versioning \
  --bucket $BUCKET \
  --versioning-configuration Status=Enabled

aws s3api get-bucket-versioning --bucket $BUCKET

#Disable Block public access
aws s3api put-public-access-block \
  --bucket $BUCKET \
  --public-access-block-configuration \
  BlockPublicAcls=false,IgnorePublicAcls=false,BlockPublicPolicy=false,RestrictPublicBuckets=false


aws iam create-user --user-name velero

aws iam list-users

cat <<EOF > velero-policy.json 
{
    "Version": "2012-10-17",
    "Statement": [
        {
            "Effect": "Allow",
            "Action": [
                "ec2:DescribeVolumes",
                "ec2:DescribeSnapshots",
                "ec2:CreateTags",
                "ec2:CreateVolume",
                "ec2:CreateSnapshot",
                "ec2:DeleteSnapshot"
            ],
            "Resource": "*"
        },
        {
            "Effect": "Allow",
            "Action": [
                "s3:GetObject",
                "s3:DeleteObject",
                "s3:PutObject",
                "s3:PutObjectTagging",
                "s3:AbortMultipartUpload",
                "s3:ListMultipartUploadParts"
            ],
            "Resource": [
                "arn:aws:s3:::${BUCKET}/*"
            ]
        },
        {
            "Effect": "Allow",
            "Action": [
                "s3:ListBucket"
            ],
            "Resource": [
                "arn:aws:s3:::${BUCKET}"
            ]
        }
    ]
}
EOF

aws iam put-user-policy \
  --user-name velero \
  --policy-name velero \
  --policy-document file://velero-policy.json

aws iam create-access-key --user-name velero

# The result should look like:

# {
#   "AccessKey": {
#         "UserName": "velero",
#         "Status": "Active",
#         "CreateDate": "2017-07-31T22:24:41.576Z",
#         "SecretAccessKey": <AWS_SECRET_ACCESS_KEY>,
#         "AccessKeyId": <AWS_ACCESS_KEY_ID>
#   }
# }

cat > credentials-velero <<EOF
[default]
aws_access_key_id=<AWS_ACCESS_KEY_ID>
aws_secret_access_key=<AWS_SECRET_ACCESS_KEY>
EOF

# Install Velero, including all prerequisites, into the cluster and start the deployment. This will create a namespace called velero, and place a deployment named velero in it.

# If using IAM user and access key:

velero install \
    --provider aws \
    --plugins velero/velero-plugin-for-aws:v1.13.0 \
    --bucket $BUCKET \
    --backup-location-config region=$REGION \
    --snapshot-location-config region=$REGION \
    --secret-file ./credentials-velero

kubectl logs deployment/velero -n velero

kubectl get all -n velero

kubectl create ns dev
kubectl get ns

kubectl apply -n dev -f -<<EOF
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx-deployment
  labels:
    app: nginx
spec:
  replicas: 3
  selector:
    matchLabels:
      app: nginx
  template:
    metadata:
      labels:
        app: nginx
    spec:
      containers:
      - name: nginx
        image: nginx:1.14.2
        ports:
        - containerPort: 80
EOF

kubectl get all -n dev

velero backup create dev-backup --include-namespaces dev

kubectl get ns
kubectl delete ns dev

velero get backups

velero restore create --from-backup dev-backup

velero restore describe dev-backup-<id>

velero restore logs dev-backup-<id>

kubectl get ns

kubectl get all -n dev

#NORMAL BUCKET
# Empty the bucket
aws s3 rm s3://$BUCKET --recursive

# Delete the bucket
aws s3 rb s3://$BUCKET
#or
aws s3 rb s3://$BUCKET --force

#If Bucket Has Versioning Enabled
aws s3api delete-objects \
  --bucket $BUCKET \
  --delete "$(aws s3api list-object-versions \
    --bucket $BUCKET \
    --query='{Objects: DeleteMarkers[].{Key:Key,VersionId:VersionId}}' \
    --output json)"

aws s3api delete-objects \
  --bucket $BUCKET \
  --delete "$(aws s3api list-object-versions \
    --bucket $BUCKET \
    --query='{Objects: Versions[].{Key:Key,VersionId:VersionId}}' \
    --output json)"

aws s3 rb s3://$BUCKET
```
