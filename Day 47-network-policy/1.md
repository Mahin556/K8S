* By default in kubernetes all pod can communicate with each other without any restrictions.

#### Example
```bash
k create ns np-demo

k apply -f -<<EOF
apiVersion: v1
kind: Pod
metadata:
  name: pod1
  labels:
    app: nginx-pod1
  namespace: np-demo
spec:
  containers:
    - name: nginx-container
      image: nginx:latest
      ports:
        - containerPort: 80
---

apiVersion: v1
kind: Service
metadata:
  name: pod1-service
spec:
  selector:
    app: nginx-pod1
  type: ClusterIP 
  ports:
    - protocol: TCP
      port: 80 
      targetPort: 80 

EOF

k apply -f -<<EOF
apiVersion: v1
kind: Pod
metadata:
  name: pod2
  labels:
    app: nginx-pod2
  namespace: np-demo
spec:
  containers:
    - name: nginx-container
      image: nginx:latest
      ports:
        - containerPort: 80
---

apiVersion: v1
kind: Service
metadata:
  name: pod2-service
spec:
  selector:
    app: nginx-pod2
  type: ClusterIP 
  ports:
    - protocol: TCP
      port: 80 
      targetPort: 80 
EOF

k config set-context --current --namespace np-demo

k get pod
NAME   READY   STATUS    RESTARTS   AGE
pod1   1/1     Running   0          50s
pod2   1/1     Running   0          49s

k exec -it pod1 -- /bin/sh -c "apt update && apt install telnet -y"
k exec -it pod2 -- /bin/sh -c "apt update && apt install telnet -y"

k exec -it pod1 -- /bin/sh -c "telnet pod2-service 80"
k exec -it pod1 -- /bin/sh -c "telnet pod1-service 80"
```

* Applying Network Policy to restrict the communication between `pod1` and `pod2`.

#### Example
```bash
# Pod1 can only receive packets on port 80 from pods having label(app: nginx-pod2)
k apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: pod1-np                
  namespace: np-demo             
spec:
  policyTypes:
    - Ingress                   
    - Egress                    
  podSelector:
    matchLabels:
      app: nginx-pod1                 
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: nginx-pod2  
      ports:
        - protocol: TCP
          port: 80           
EOF

k exec -it pod2 -- /bin/sh -c "telnet pod1-service 80" #Work✅

k exec -it pod1 -- /bin/sh -c "telnet pod2-service 80" #No Work❌


k apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: pod2-np                
  namespace: np-demo             
spec:
  policyTypes:
    - Ingress                   
    - Egress                    
  podSelector:
    matchLabels:
      app: nginx-pod2                 
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: nginx-pod1  
      ports:
        - protocol: TCP
          port: 80           
EOF

k exec -it pod2 -- /bin/sh -c "telnet pod1-service 80" #No Work❌

k exec -it pod1 -- /bin/sh -c "telnet pod2-service 80" #No Work❌


k apply -f -<<EOF
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: demo                
  namespace: np-demo             
spec:
  policyTypes:
    - Ingress                   
    - Egress

    matchLabels:
      app: nginx-pod2                         
EOF
```
